	public int DeleteTran(int Code)
	{
		int result = 0;
		try
		{
			int TranID = ((IQueryable<Tran_Tran>)db.Tran_Tran).Where((Tran_Tran x) => x.ReferenceCode == (int?)Code && x.TranTypeID == (int?)5 && x.BranchID == (int?)LoginDetails.BranchID).FirstOrDefault().ID;
			Tran_Tran tran_Tran = ((IQueryable<Tran_Tran>)db.Tran_Tran).Where((Tran_Tran x) => x.ID == TranID && x.BranchID == (int?)LoginDetails.BranchID).FirstOrDefault();
			if (tran_Tran != null)
			{
				db.Tran_Tran.Attach(tran_Tran);
				db.Tran_Tran.Remove(tran_Tran);
				result = ((DbContext)db).SaveChanges();
			}
		}
		catch (Exception)
		{
		}
		return result;
	}

	public int DeleteTranDetails(int Code)
	{
		int result = 0;
		try
		{
			int TranID = ((IQueryable<Tran_Tran>)db.Tran_Tran).Where((Tran_Tran x) => x.ReferenceCode == (int?)Code && x.TranTypeID == (int?)5 && x.BranchID == (int?)LoginDetails.BranchID).FirstOrDefault().ID;
			List<Tran_TranDetails> list = ((IQueryable<Tran_TranDetails>)db.Tran_TranDetails).Where((Tran_TranDetails x) => x.TranSn == (int?)TranID && x.BranchID == (int?)LoginDetails.BranchID).ToList();

				foreach (Tran_TranDetails item in list)

				}

				}
				break;
			}
			case "الكبرى":
			{
				ItemQuantity itemQuantity = ((IQueryable<ItemQuantity>)db.ItemQuantities).Where((ItemQuantity x) => x.ItemID == (int?)ItemID && x.StoreID == (int?)StoreID).FirstOrDefault();

					decimal num = Convert.ToDecimal(item_Items.ConvertMediumUnit);

			}
			}
